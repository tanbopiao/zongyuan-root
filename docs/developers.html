💻 API调用示例
# Python 示例 - 调用健康检查API
import requests

# 配置API端点和密钥
base_url = "https://www.huodouai.com/api/v1"
api_key = "your-api-key-here"

# 设置请求头
headers = {
    "Authorization": f"Bearer {api_key}",
    "Content-Type": "application/json"
}

# 调用健康检查API
response = requests.get(f"{base_url}/health", headers=headers)
print(response.json())

# 调用合规校验API
data = {"content": "待校验内容", "rule_ids": ["R001", "R002"]}
response = requests.post(f"{base_url}/rules/check", json=data, headers=headers)
print(response.json())
